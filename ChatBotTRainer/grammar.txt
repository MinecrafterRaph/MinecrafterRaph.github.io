# Natural language grammar templates and simple correction rules
# This file is used by the chatbot to build more connected sentences.

subject:Ich|Du|Er|Sie|Wir|Ihr
verb:kenne|lerne|verstehe|mag|habe|möchte|kann|sage|denke|sehe|finde
object:Mathe|Grammatik|Code|Wörter|Fragen|Antworten|Sprache|Satzbau|Logik|Musik|Zahlen|Rechnen
adjective:gut|einfach|spannend|besser|komplex|interessant|klar|nützlich|schnell
connector:und|aber|denn|deshalb|oder|währenddessen
adverb:gerne|jetzt|oft|manchmal|ganz|wirklich

template:{subject} {verb} {object}.
template:{subject} {verb} {object} {connector} ich {verb} {object}.
template:Ich finde {object} {adjective}.
template:Das ist {adjective} {connector} sehr wichtig.
template:Ich {verb} {object} {adverb}.
template:Manchmal ist {object} {adjective}.
template:Ich möchte dir bei {object} helfen.

template:Wenn du willst, kann ich mehr über {object} erzählen.

template:Das klingt nach {object} {connector} ich {verb} {object}.

template:Ich versuche, dir bei {object} mit etwas {adjective}em zu helfen.

correct:ich gehe -> Ich gehe
correct:du ist -> du bist
correct:ich habe nicht -> Ich habe nicht
correct:das ist gut -> Das ist gut
correct:wir sind freundlich -> Wir sind freundlich
